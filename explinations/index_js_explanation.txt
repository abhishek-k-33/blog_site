--- 1. getAllPosts() (Lines 123 - 139) ---
Lines 123-136:
    const getAllPosts = async () => {
        if (supabase) {
            try {
                const { data, error } = await supabase
                    .from("posts")
                    .select("*")
                    .order("created_at", { ascending: false });
                if (error) throw error;
                return (data || []).map(formatPost);
            } catch (err) {
                console.error("Supabase getAllPosts error:", err.message);
                return [];
            }
        }
        const localPosts = await readLocalPosts();
        return localPosts.map(formatPost);
    };
    • Queries all posts from Supabase ordered by newest first ('created_at' DESC).
    • If Supabase is offline or not configured, falls back to reading 'data.json'.


--- 2. getPostById(id) (Lines 141 - 159) ---
Lines 141-159:
    const getPostById = async (id) => {
        if (supabase) {
            try {
                const { data, error } = await supabase
                    .from("posts")
                    .select("*")
                    .eq("id", id)
                    .maybeSingle();
                if (error) throw error;
                return data ? formatPost(data) : null;
            } catch (err) {
                console.error("Supabase getPostById error:", err.message);
                return null;
            }
        }
        const localPosts = await readLocalPosts();
        const post = localPosts.find((p) => String(p.id) === String(id));
        return post ? formatPost(post) : null;
    };
    • Fetches a single post matching the given ID from Supabase or local array.
